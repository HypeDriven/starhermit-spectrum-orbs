# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com by hash; never edit it). Re-run `npm run package`
# after every change and commit index.html + starhermit-files.json.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.30.0
engine.minor=32
engine.capabilities=platform,storage,clipboard

# Keyboard actions (players rebind keys and pads in the runtime's Options).
# The gallery is also played with the pointer or a finger: tap a tube (or its
# number, or its chip in the board-status strip) to lift its top orb, tap
# another to pour it.
engine.group.prev=Tubes
control.prev=ArrowLeft+ArrowUp | Previous tube
control.next=ArrowRight+ArrowDown | Next tube
control.confirm=Enter+Space | Select / pour
control.cancel=Escape | Cancel / close
engine.group.pause=Round
control.pause=KeyP | Pause
control.undo=KeyU | Undo
control.hint=KeyH | Hint
control.restart=KeyR | Restart board
control.camera=KeyC | Reset camera

# Gamepad: the browser build's defaults (A confirm, B cancel, X undo, Y hint,
# Start pause; D-pad or left stick moves the tube focus), plus Select to
# restart and the right stick press to reset the camera.
engine.gamepad.prev=DpadLeft+DpadUp+LeftStickLeft
engine.gamepad.next=DpadRight+DpadDown+LeftStickRight
engine.gamepad.confirm=South
engine.gamepad.cancel=East
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.hint=North
engine.gamepad.restart=Select
engine.gamepad.camera=RightStickPress
